const NOME_ABA = 'Registros (ao vivo)';
const CABECALHO = ['Data', 'Horario', 'Tipo', 'Aluno', 'Matricula', 'Turma', 'Local', 'Telefone', 'Motivo', 'Autorizado/Registrado por', 'Última atualização'];

/**
 * Função principal — chamada automaticamente pelo gatilho de tempo.
 * Também pode ser rodada manualmente pelo menu Extensões > Apps Script > Executar.
 */
function sincronizarRegistros() {
  const idToken = obterIdToken_();
  const documentos = buscarTodosOsRegistros_(idToken);
  const registros = converterDocumentos_(documentos);
  escreverNaPlanilha_(registros);
  escreverResumoPorAluno_(registros);
}

/**
 * Faz login com o usuário "robô" via API do Firebase Authentication
 * (REST) e retorna um idToken válido por 1 hora, usado para autenticar
 * as chamadas ao Firestore.
 */
function obterIdToken_() {
  const url = `https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=${FIREBASE_API_KEY}`;
  const resposta = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({
      email: EMAIL_SCRIPT,
      password: SENHA_SCRIPT,
      returnSecureToken: true
    }),
    muteHttpExceptions: true
  });

  const dados = JSON.parse(resposta.getContentText());
  if (!dados.idToken) {
    throw new Error('Falha ao autenticar no Firebase: ' + resposta.getContentText());
  }
  return dados.idToken;
}

/**
 * Busca todos os documentos da coleção "registros" no Firestore,
 * paginando automaticamente (a API do Firestore devolve no máximo
 * 300 documentos por chamada).
 */
function buscarTodosOsRegistros_(idToken) {
  let documentos = [];
  let pageToken = null;

  do {
    let url = `https://firestore.googleapis.com/v1/projects/${FIREBASE_PROJECT_ID}/databases/(default)/documents/registros?pageSize=300`;
    if (pageToken) url += `&pageToken=${pageToken}`;

    const resposta = UrlFetchApp.fetch(url, {
      method: 'get',
      headers: { Authorization: 'Bearer ' + idToken },
      muteHttpExceptions: true
    });

    const dados = JSON.parse(resposta.getContentText());
    if (dados.error) {
      throw new Error('Erro do Firestore: ' + JSON.stringify(dados.error));
    }
    if (dados.documents) documentos = documentos.concat(dados.documents);
    pageToken = dados.nextPageToken || null;
  } while (pageToken);

  return documentos;
}

/**
 * Remove acentos e força maiúscula — usada para AGRUPAR/COMPARAR nomes
 * (mesma regra usada no app: "João" e "Joao" contam como o mesmo aluno).
 */
function chaveNormalizada_(texto) {
  return (texto || '').toString().trim().normalize('NFD').replace(/[\u0300-\u036f]/g, '').toUpperCase();
}

function paraMaiuscula_(texto) {
  return (texto || '').toString().toUpperCase();
}

/**
 * Identifica um aluno de forma confiável: nome completo + turma
 * (nunca só o primeiro nome, para não misturar alunos diferentes
 * com o mesmo nome).
 */
function chaveDoAluno_(reg) {
  return chaveNormalizada_(reg.aluno) + '||' + chaveNormalizada_(reg.turma);
}

/**
 * Converte os documentos crus do Firestore num array de objetos
 * simples, já prontos para uso tanto na aba de registros quanto na
 * aba de resumo.
 */
function converterDocumentos_(documentos) {
  return documentos.map(doc => {
    const f = doc.fields || {};
    return {
      data: extrairValor_(f.data),
      horario: extrairValor_(f.horario),
      tipo: extrairValor_(f.tipo),
      aluno: extrairValor_(f.aluno),
      matricula: extrairValor_(f.matricula),
      turma: extrairValor_(f.turma),
      local: extrairValor_(f.local),
      telefone: extrairValor_(f.telefone),
      motivo: extrairValor_(f.motivo),
      autorizado: extrairValor_(f.autorizado)
    };
  });
}

/**
 * Converte o formato de campos do Firestore (ex: {stringValue: "x"})
 * para um valor JS simples.
 */
function extrairValor_(campo) {
  if (!campo) return '';
  if ('stringValue' in campo) return campo.stringValue;
  if ('integerValue' in campo) return Number(campo.integerValue);
  if ('doubleValue' in campo) return Number(campo.doubleValue);
  if ('timestampValue' in campo) return campo.timestampValue;
  if ('booleanValue' in campo) return campo.booleanValue;
  return '';
}

function escreverNaPlanilha_(registros) {
  const planilha = SpreadsheetApp.openById(ID_PLANILHA_DESTINO);
  let aba = planilha.getSheetByName(NOME_ABA);
  if (!aba) aba = planilha.insertSheet(NOME_ABA);

  aba.clearContents();
  aba.appendRow(CABECALHO);

  const agora = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss');

  const linhas = registros.map(reg => [
    reg.data, reg.horario, paraMaiuscula_(reg.tipo), paraMaiuscula_(reg.aluno),
    paraMaiuscula_(reg.matricula), paraMaiuscula_(reg.turma), paraMaiuscula_(reg.local),
    reg.telefone, paraMaiuscula_(reg.motivo), paraMaiuscula_(reg.autorizado), agora
  ]);

  if (linhas.length > 0) {
    aba.getRange(2, 1, linhas.length, CABECALHO.length).setValues(linhas);
  }

  aba.getRange(1, 1, 1, CABECALHO.length).setFontWeight('bold');
  aba.autoResizeColumns(1, CABECALHO.length);
}

/**
 * Aba "Resumo por Aluno" — mesma lógica usada no botão "Exportar XLSX"
 * do app: contagem diária por aluno+turma+tipo, e um total geral por
 * aluno (todos os tipos somados), ignorando acento e maiúscula/minúscula.
 */
function escreverResumoPorAluno_(registros) {
  const planilha = SpreadsheetApp.openById(ID_PLANILHA_DESTINO);
  const NOME_ABA_RESUMO = 'Resumo por Aluno';
  let aba = planilha.getSheetByName(NOME_ABA_RESUMO);
  if (!aba) aba = planilha.insertSheet(NOME_ABA_RESUMO);
  aba.clearContents();

  // Tabela 1: contagem por aluno+turma+data+tipo
  const contagemPorDia = {};
  registros.forEach(reg => {
    const chave = chaveDoAluno_(reg) + '||' + reg.data + '||' + reg.tipo;
    if (!contagemPorDia[chave]) {
      contagemPorDia[chave] = { nome: paraMaiuscula_(reg.aluno), turma: paraMaiuscula_(reg.turma), data: reg.data, tipo: paraMaiuscula_(reg.tipo), quantidade: 0 };
    }
    contagemPorDia[chave].quantidade++;
  });
  const tabela1 = Object.keys(contagemPorDia).map(k => contagemPorDia[k]);

  // Tabela 2: total geral por aluno, somando todos os tipos
  const totalGeral = {};
  registros.forEach(reg => {
    const chave = chaveDoAluno_(reg);
    if (!totalGeral[chave]) {
      totalGeral[chave] = { nome: paraMaiuscula_(reg.aluno), turma: paraMaiuscula_(reg.turma), total: 0 };
    }
    totalGeral[chave].total++;
  });
  const tabela2 = Object.keys(totalGeral).map(k => totalGeral[k]).sort((a, b) => b.total - a.total);

  // Tabela 1: colunas A-E
  aba.getRange(1, 1, 1, 5).setValues([['Aluno', 'Turma', 'Data', 'Tipo', 'Quantidade']]);
  if (tabela1.length > 0) {
    const linhas1 = tabela1.map(item => [item.nome, item.turma, item.data, item.tipo, item.quantidade]);
    aba.getRange(2, 1, linhas1.length, 5).setValues(linhas1);
  }

  // Tabela 2: colunas G-I (coluna F fica em branco como espaçador)
  aba.getRange(1, 7, 1, 3).setValues([['Aluno', 'Turma', 'Total Geral (todos os tipos)']]);
  if (tabela2.length > 0) {
    const linhas2 = tabela2.map(item => [item.nome, item.turma, item.total]);
    aba.getRange(2, 7, linhas2.length, 3).setValues(linhas2);
  }

  aba.getRange(1, 1, 1, 5).setFontWeight('bold');
  aba.getRange(1, 7, 1, 3).setFontWeight('bold');
  aba.autoResizeColumns(1, 9);
}

/**
 * Roda esta função UMA VEZ (manualmente) para criar o gatilho automático
 * que chama sincronizarRegistros() a cada 15 minutos, sozinho, para sempre.
 */
function configurarAtualizacaoAutomatica() {
  // Remove gatilhos antigos deste script, se existirem, para não duplicar
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'sincronizarRegistros') ScriptApp.deleteTrigger(t);
  });

  ScriptApp.newTrigger('sincronizarRegistros')
    .timeBased()
    .everyMinutes(15)
    .create();

  Logger.log('Atualização automática configurada: a cada 15 minutos.');
}
